// Platform-specific AST transformation engine
export class HashnodeAdapter implements PublishingPlatform {
  async transform(article: Article): Promise<PlatformPayload> {
    const ast = unified()
      .use(remarkParse)
      .use(remarkGfm)
      .use(transformCodeBlocksToGistOrHighlight)
      .use(resolveLocalCdnImages, { cdn: 'https://cdn.artxflow.dev' })
      .use(remarkStringify);

    return {
      publicationId: env.HASHNODE_PUB_ID,
      input: {
        title: article.title,
        contentMarkdown: String(await ast.process(article.content)),
        canonicalUrl: article.canonicalUrl, // Strict SEO
        tags: article.tags.map(t => ({ slug: t.slug, name: t.name }))
      }
    };
  }
}
